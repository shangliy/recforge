% main loop
\node[box, fill=cMeta!15, minimum width=27mm] (prop) {\textbf{Proposer}\\LLM agent or search\\[1pt]\scriptsize reads archive, memory, skills};
\node[box, fill=cL2!12, right=6mm of prop, minimum width=34mm] (cand) {\textbf{Candidate change}\\[1pt]\scriptsize
  \textcolor{cL0!70!black}{L0} space point \;

  \textcolor{cL2}{L2} code\\\scriptsize

\node[box, fill=gray!8, right=6mm of cand, minimum width=27mm] (train) {\textbf{Train and evaluate}\\[1pt]\scriptsize data splits, metric,\\\scriptsize evaluation code};
\node[box, fill=cL1!12, right=6mm of train, minimum width=31mm] (sel) {\textbf{Selection signal}\\[1pt]\scriptsize S1 val + sealed test \; S2 test\\\scriptsize S3 unstated \; S4 proxy\\\scriptsize S5 online A/B \; S6 self-feedback};
\node[box, fill=cMeta!15, below=11mm of cand, minimum width=34mm] (arch) {\textbf{Archive / incumbent}\\\scriptsize kept candidates become the next baseline};
\node[box, fill=gray!8, below=11mm of sel.south east, anchor=north east, xshift=9mm, minimum width=24mm] (out) {\textbf{Report or launch}\\\scriptsize test once \;/\; online experiment};

\caption{Anatomy of a self-evolving recommender. A proposer changes some part of the recommender (the evolved
object, axis 1 of the taxonomy); the changed system is trained and evaluated; a selection signal (axis 2) decides
what is kept. Red markers show where the evidence for a reported gain can fail, with the audit criterion that checks it:
\textbf{1} the agent edits evaluation code, reads test data or games the metric (C9);
\textbf{2} sampled or unstated ranking metrics (C1);
\textbf{3} the maximum over many candidates on one reused split, or selection on test (C2, C3);
\textbf{4} a proxy or self-generated signal that does not transfer (S4, S6);
\textbf{5} a gain inside run-to-run noise becomes the new baseline (C4, C5);
\textbf{6} short or repeated online experiments without correction for multiple testing;
\textbf{7} gains measured against an untuned starting point, with no equal-budget null search (C6, C8).}
\label{fig:loop}
\end{figure*}
